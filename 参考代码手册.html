模块A · 基础入门

BeeBehaviour.cs 基础 · 生命周期与对象信息

所有脚本都继承 MonoBehaviour。Start 在场景加载后调用一次，Update 每帧调用。Debug.Log 在控制台打印，transform/gameObject 读取对象本身。

using UnityEngine;

public class BeeBehaviour : MonoBehaviour
{
    // Start：场景加载后、第一次 Update 前调用一次，适合初始化
    void Start()
    {
        Debug.Log("蜜蜂启动！");

        // gameObject / this：拿到脚本所在的游戏对象
        string beeName = this.gameObject.name;
        Debug.Log("【蜜蜂】所属模型：" + beeName);

        // transform：拿到变换组件，读取三维坐标
        Vector3 position = this.transform.position;
        Debug.Log("【蜜蜂】三维坐标：" + position.ToString("F3"));
    }

    // Update：每一帧调用一次，大部分游戏逻辑写在这里
    void Update()
    {
        Debug.Log("【蜜蜂】当前时间：" + Time.time);
        Debug.Log("【蜜蜂】距上次帧间隔：" + Time.deltaTime);
    }
}
BNUTransR.cs 基础 · 旋转 / 位移 / 匀速运动

Transform.Rotate 绕轴旋转，Transform.Translate 沿相对方向位移。乘 Time.deltaTime 后变为与帧率无关的匀速运动。

using UnityEngine;

public class BNUTransR : MonoBehaviour
{
    void Update()
    {
        // 不乘 deltaTime：受帧率影响，非匀速
        this.transform.Rotate(2, 0, 0);                  // 每帧绕 X 轴转 2°
        this.transform.Translate(0, 0, 1);               // 每帧沿 Z 轴移 1 单位

        // 乘 deltaTime：匀速（推荐）
        this.transform.Rotate(10 * Time.deltaTime, 0, 0);
        this.transform.Translate(0, 0, 1 * Time.deltaTime);
    }
}
BeePingPong.cs 基础 · 往返运动（作业参考）

在上下边界之间来回运动。用 Translate 做相对位移，比直接改 position 更清晰。

using UnityEngine;

public class BeePingPong : MonoBehaviour
{
    public float top = 3f;       // 上边界
    public float bottom = -3f;   // 下边界
    public float speed = 2f;     // 速度
    private int dir = 1;         // 方向：1 向上，-1 向下

    void Update()
    {
        float z = transform.position.z;
        if (z > top) dir = -1;
        else if (z < bottom) dir = 1;

        transform.Translate(0, 0, dir * speed * Time.deltaTime);
    }
}

模块B · 键盘控制与受力

BeeMove.cs 键盘移动（Input.GetAxis 优化版）

用 Input.GetAxis 拿到 -1~1 的平滑输入，比一堆 if GetKey 简洁。VerticalY 是自定义轴线（输入管理器里 Name=VerticalY，Alt Negative=K，Alt Positive=J）。

using UnityEngine;

public class BeeMove : MonoBehaviour
{
    public float moveSpeed = 6f;

    void Update()
    {
        float horizontal = Input.GetAxis("Horizontal");   // A/D 或 ←/→
        float vertical   = Input.GetAxis("Vertical");     // W/S 或 ↑/↓
        float verticalY  = Input.GetAxis("VerticalY");    // 自定义：J/K 升降

        Vector3 move = new Vector3(horizontal, verticalY, vertical);
        transform.Translate(move * moveSpeed * Time.deltaTime);
    }
}
BeeJump.cs 受力跳跃（已修正）

【修正 2025 PPT 的 bug】原 PPT 末尾多写了 '}}' 导致编译错误，已删除。水平移动用运动学 Translate，跳跃用刚体 AddForce 冲量。

using UnityEngine;

public class BeeJump : MonoBehaviour
{
    private Rigidbody rb;
    public float jumpForce = 5f;
    public float moveSpeed = 6f;

    void Start()
    {
        rb = GetComponent<Rigidbody>();
    }

    void Update()
    {
        float horizontal = Input.GetAxis("Horizontal");
        float vertical   = Input.GetAxis("Vertical");
        float verticalY  = Input.GetAxis("VerticalY");
        Vector3 move = new Vector3(horizontal, verticalY, vertical);
        transform.Translate(move * moveSpeed * Time.deltaTime);

        if (Input.GetButtonDown("Jump"))
        {
            rb.AddForce(Vector3.up * jumpForce, ForceMode.Impulse);
        }
    }
}
ForceMove.cs 小球受力滚动（已修正）

【修正 2025 PPT 的 bug】原 PPT 末尾多写了 '}}' 导致编译错误，已删除。每帧施加方向力，小球滚动。

模块C · FPS 玩家控制

PlayerMove.cs 角色控制器移动

用 CharacterController 而非刚体，移动可控、不穿墙。transform.right/forward 跟随玩家朝向。

using UnityEngine;

public class PlayerMove : MonoBehaviour
{
    public CharacterController controller;
    public float moveSpeed = 10f;

    void Start()
    {
        controller = GetComponent<CharacterController>();
    }

    void Update()
    {
        float horizontal = Input.GetAxis("Horizontal"); // A/D
        float vertical   = Input.GetAxis("Vertical");   // W/S
        Vector3 move = transform.right * horizontal + transform.forward * vertical;
        controller.Move(move * moveSpeed * Time.deltaTime);
    }
}
PlayerRotate.cs 鼠标控制视角

身体（playerTrans）绕 Y 轴左右转，摄像机（主摄像机为玩家子物体）绕 X 轴上下俯仰并限制角度，避免翻转。

using UnityEngine;

public class PlayerRotate : MonoBehaviour
{
    public Transform playerTrans;       // 玩家（身体）Transform
    public Transform mainCameraTrans;   // 主摄像机 Transform（挂载为玩家子物体）
    public float rotateSpeed = 180f;
    [Range(2f, 4f)]
    public float rotateRatio = 3f;      // 旋转敏捷度

    private float xRotateOffset;        // 摄像机俯仰偏移

    void Update()
    {
        float mouseX = Input.GetAxis("Mouse X");
        float mouseY = Input.GetAxis("Mouse Y");

        playerTrans.Rotate(Vector3.up * mouseX * rotateSpeed * rotateRatio * Time.deltaTime);

        xRotateOffset -= mouseY * rotateSpeed * rotateRatio * Time.deltaTime;
        xRotateOffset = Mathf.Clamp(xRotateOffset, -60f, 60f);

        mainCameraTrans.localRotation = Quaternion.Euler(xRotateOffset, 0f, 0f);
    }
}
PlayerJump.cs 角色控制器跳跃（补充课件13）

CharacterController 没有自带重力，需自己维护竖直速度。用 isGrounded 判断在地面，否则会无限跳。

using UnityEngine;

public class PlayerJump : MonoBehaviour
{
    public CharacterController controller;
    public float moveSpeed = 10f;
    public float jumpHeight = 2f;
    public float gravity = -9.81f;

    private Vector3 velocity;

    void Update()
    {
        if (controller.isGrounded && velocity.y < 0)
            velocity.y = -2f;

        float h = Input.GetAxis("Horizontal");
        float v = Input.GetAxis("Vertical");
        Vector3 move = transform.right * h + transform.forward * v;
        controller.Move(move * moveSpeed * Time.deltaTime);

        if (Input.GetButtonDown("Jump") && controller.isGrounded)
            velocity.y = Mathf.Sqrt(jumpHeight * -2f * gravity);

        velocity.y += gravity * Time.deltaTime;
        controller.Move(velocity * Time.deltaTime);
    }
}
ShootingMouseLock.cs 锁定鼠标

FPS 必须把鼠标锁在屏幕中心，否则点出窗口就打断射击。按 ESC 可解锁（调试/菜单用）。

using UnityEngine;

public class ShootingMouseLock : MonoBehaviour
{
    private bool isCursorLocked = true;

    void Start()
    {
        LockCursor();
    }

    void Update()
    {
        if (Input.GetKeyDown(KeyCode.Escape))
        {
            isCursorLocked = !isCursorLocked;
            if (isCursorLocked) LockCursor();
            else UnlockCursor();
        }
    }

    private void LockCursor()
    {
        Cursor.visible = false;
        Cursor.lockState = CursorLockMode.Locked; // 锁定在中心，移出窗口也捕获输入
        isCursorLocked = true;
    }

    private void UnlockCursor()
    {
        Cursor.visible = true;
        Cursor.lockState = CursorLockMode.None;
        isCursorLocked = false;
    }
}
WeaponController.cs 射击（发射子弹）

鼠标左键 Instantiate 子弹预制体，给刚体一个向前的速度，1 秒后 Destroy。子弹预制体需取消勾选 Use Gravity。

using UnityEngine;

public class WeaponController : MonoBehaviour
{
    public Transform bulletStartPoint; // 子弹发射位置（空物体）
    public GameObject bullet;          // 子弹预制体
    public float bulletSpeed = 50f;    // 子弹速度

    void Update()
    {
        OpenFire();
    }

    private void OpenFire()
    {
        if (Input.GetMouseButtonDown(0)) // 鼠标左键
        {
            GameObject newBullet = Instantiate(bullet, bulletStartPoint.position, bulletStartPoint.rotation);
            newBullet.GetComponent<Rigidbody>().velocity = newBullet.transform.forward * bulletSpeed;
            Destroy(newBullet, 1f);
        }
    }
}

模块D · 敌人与战斗

EnemyControl.cs 敌人跟踪玩家（NavMesh，已修正）

【修正 2025 PPT 的 bug】原 PPT 把 enemyAgent.destination 写在了 Update 大括号外面，导致编译错误，已放进 Update 内。需先 Bake NavMesh 并给敌人加 Nav Mesh Agent。

using UnityEngine;
using UnityEngine.AI;

public class EnemyControl : MonoBehaviour
{
    private NavMeshAgent enemyAgent;
    private GameObject player;

    void Start()
    {
        enemyAgent = GetComponent<NavMeshAgent>();
        player = GameObject.FindGameObjectWithTag("Player");
    }

    void Update()
    {
        // 寻路目标 = 玩家当前位置
        enemyAgent.destination = player.transform.position;
    }
}
EnemyData.cs 敌人攻击参数（距离判定 + 主动攻击 + 咬的动作）

敌人够近（attackRange）+ 冷却好了（attackInterval）就主动调玩家的 TakeDamage。Start 里用 Player 标签找一次玩家。挂动物预制体的空壳上，攻击力 10 / 间隔 1 / 范围 2.2。咬的瞬间调 AnimalAnim.PlayAttack() 播一下咬的动作（动画参数 Attack 触发器在课件18 配的，没配的动物不带动静）。
⛔ 不要改回 OnControllerColliderHit + CompareTag("Enemy") 的物理接触判定：它只在「玩家自己移动撞到东西」时响，玩家站着不动、动物蹭过来不响；跟随的动物又停在离玩家 2 米处，物理上碰不到，实测扣血极稀疏。

using UnityEngine;

public class EnemyData : MonoBehaviour
{
    public float attackPower = 10f;      // 每次咬掉多少血
    public float attackInterval = 1f;    // 两次攻击至少隔几秒
    public float attackRange = 2.2f;     // 离玩家多近算咬到

    private float lastAttackTime;        // 上一次攻击的时刻
    private HealthChange player;         // 玩家身上的扣血脚本
    private AnimalAnim anim;             // 动物身上的动画脚本

    void Start()
    {
        // 开局找一次玩家（Find 比较慢，别放到 Update 里每帧找）
        GameObject find = GameObject.FindGameObjectWithTag("Player");
        if (find != null) player = find.GetComponent<HealthChange>();

        // 动画脚本就在自己身上（课件18 挂的 AnimalAnim）
        anim = GetComponent<AnimalAnim>();
    }

    void Update()
    {
        if (player == null) return;      // 找不到玩家就什么都不做，避免报空引用

        // 只在水平面上量距离：高度差不算，上下坡才不会误判
        Vector3 me = transform.position;          me.y = 0f;
        Vector3 you = player.transform.position;  you.y = 0f;

        // 够近 + 冷却好了 → 咬一口
        if (Vector3.Distance(me, you) <= attackRange
            && Time.time - lastAttackTime >= attackInterval)
        {
            lastAttackTime = Time.time;
            anim.PlayAttack();           // 先动起来，再让玩家掉血
            player.TakeDamage(attackPower);
        }
    }
}
HealthChange.cs 玩家血量与血条

玩家只负责三件事：开局满血、被咬扣血、把血量换算成 fillAmount。不自己检测碰撞——攻击由 EnemyData 发起，所以 TakeDamage 是 public。挂载在 Player 上，FillImageHealth 拖入生命条填充图。
谁咬就扣谁那一口，多只同时命中会叠加（频率由敌人各自的攻击间隔管住）；血量归零后不再吃伤害。

using UnityEngine;
using UnityEngine.UI;   // 要用 Image，必须带上这一行

public class HealthChange : MonoBehaviour
{
    public float MaxHealth = 100f;          // 满血
    public float CurrentHealth;             // 当前血量
    public Image healthFillImage;           // 血条上那条红条，检查器里拖进来

    void Start()
    {
        CurrentHealth = MaxHealth;          // 开局满血
        UpdateHealth();
    }

    // 扣血：敌人咬到时会调这个方法，扣完立刻刷新血条
    public void TakeDamage(float damage)
    {
        // 血已经空了，就不再吃伤害（动物还在咬也没用）
        if (CurrentHealth <= 0) return;

        CurrentHealth = CurrentHealth - damage;
        if (CurrentHealth < 0) CurrentHealth = 0;
        UpdateHealth();
    }

    // 把血量换算成 0~1，交给红条
    void UpdateHealth()
    {
        if (healthFillImage != null)
        {
            healthFillImage.fillAmount = CurrentHealth / MaxHealth;
        }
    }
}
EnemyAttacked.cs 子弹击杀敌人（课件27）

子弹（标签 Bullet）碰到敌人即销毁敌人，并广播敌人死亡事件，通知怪物笼补充新敌人。

using UnityEngine;

public class EnemyAttacked : MonoBehaviour
{
    void OnCollisionEnter(Collision collision)
    {
        if (collision.gameObject.CompareTag("Bullet"))
        {
            Events.EnemyDeathEvent.Raise(); // 广播死亡事件
            Destroy(gameObject);            // 击杀敌人
        }
    }
}

CreateMonster.cs 敌人孵化器（怪物笼）

在出生点生成限定数量的怪物，敌人死亡后通过事件补生成，维持同屏数量。挂在 MonsterCage 上。

using UnityEngine;

public class CreateMonster : MonoBehaviour
{
    public GameObject[] monsterPrefabs; // 怪物预制体列表
    public Transform[] spawnPoints;     // 出生点（WayPoint）
    public int maxAlive = 5;            // 同屏最大数量

    private int aliveCount = 0;

    void Start()
    {
        Events.EnemyDeathEvent.OnRaise += OnEnemyDied;
        Spawn();
    }

    void Spawn()
    {
        if (aliveCount >= maxAlive) return;
        int p = Random.Range(0, monsterPrefabs.Length);
        int s = Random.Range(0, spawnPoints.Length);
        Instantiate(monsterPrefabs[p], spawnPoints[s].position, Quaternion.identity);
        aliveCount++;
    }

    void OnEnemyDied()
    {
        aliveCount--;
        Spawn();
    }

    void OnDestroy()
    {
        Events.EnemyDeathEvent.OnRaise -= OnEnemyDied;
    }
}

StartScene.cs 启动页面：按钮 + 异步加载场景（课件24）

开始按钮点击后异步加载游戏场景。场景名需与 File > Build Settings 中的名称一致。Canvas 里放 Image/Button，Button 拖入此脚本的 startButton 字段。

using UnityEngine;

public class StartScene : MonoBehaviour

    void Start()
    {
        startButton.onClick.AddListener(LoadGame);

    {
        SceneManager.LoadSceneAsync("GameScene"); // 异步加载游戏场景
    }
}

MaxOfThree.cs 求三个数的最大值（控制台）

纯 C# 控制台程序，用于熟悉变量、if 判断。在 Visual Studio 新建控制台项目运行。

using System;

class Program
{
    static void Main()
    {
        int a = 18, b = 25, c = 12;
        int max = a;
        if (b > max) max = b;
        if (c > max) max = c;
        Console.WriteLine("最大值是：" + max);
    }
}
NineNineTable.cs 九九乘法表（控制台）

嵌套 for 循环。Console.Write 不换行，\t 制表符对齐；内层结束用 Console.WriteLine 换行。

using System;

class Program
{
    static void Main()
    {
        for (int i = 1; i <= 9; i++)
        {
            for (int j = 1; j <= i; j++)
            {
                Console.Write(j + "x" + i + "=" + (i * j) + "\t");
            }
            Console.WriteLine();
        }
    }
}
